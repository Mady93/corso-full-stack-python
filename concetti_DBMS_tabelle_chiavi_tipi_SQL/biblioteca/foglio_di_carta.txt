Utenti(id INTEGER (PK AUTOINCREMENT), nome VARCHAR(50) NOT NULL, cognome VARCHAR(50) NOT NULL, email VARCHAR(50) UNIQUE NOT NULL).
Libri(id INTEGER (PK AUTOINCREMENT), titolo VARCHAR(100) NOT NULL, autore VARCHAR(100) NOT NULL, anno INTEGER).
Prestiti (id INTEGER (PK AUTOINCREMENT), id_utente INTEGER (FK - Utenti.id) NOT NULL, id_libro INTEGER (FK - Libri.id) NOT NULL, data_inizio DATE NOT NULL, data_fine DATE)
